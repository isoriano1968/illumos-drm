bp_mapout	# documented 9F (bp_mapout(9F)): unmaps a pinned user range at unpin

pf_is_memory	# non-DDI kernel interface (vm/page.h, i86pc/vm/vm_machdep.c): page_is_ram() of a frame, which amdgpu's GMC v9 checks before treating a carveout address as system memory
